cmake_minimum_required(VERSION 3.25)
project(hailstone_bastion VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
option(BASTION_SANITIZE "Address and undefined-behavior sanitizers (GCC/Clang)" OFF)
if(BASTION_SANITIZE AND NOT MSVC)
  add_compile_options(-fsanitize=address,undefined -fno-sanitize-recover=undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
set(HAILSTONE_SDK_DIR "${CMAKE_CURRENT_SOURCE_DIR}/sdk" CACHE PATH "Hailstone public SDK")
get_filename_component(_sdk_parent "${HAILSTONE_SDK_DIR}" DIRECTORY)
set(_client_ui_default "")
if(EXISTS "${_sdk_parent}/ui/gamepad/lib/window.lua")
  set(_client_ui_default "${_sdk_parent}")
endif()
set(HAILSTONE_CLIENT_UI_ROOT "${_client_ui_default}" CACHE PATH "Optional Hailstone checkout for native pack-helper UI tests")
if(HAILSTONE_CLIENT_UI_ROOT)
  foreach(helper IN ITEMS window theme pad)
    if(NOT EXISTS "${HAILSTONE_CLIENT_UI_ROOT}/ui/gamepad/lib/${helper}.lua")
      message(FATAL_ERROR "HAILSTONE_CLIENT_UI_ROOT must contain the native Gamepad pack helpers")
    endif()
  endforeach()
endif()
if(NOT EXISTS "${HAILSTONE_SDK_DIR}/hailstone_follow.h")
  message(FATAL_ERROR "Use the server-addon-services Hailstone SDK; see README.md")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(bastion_core STATIC
  src/citadel_preferences.cpp src/citadel_hub.cpp src/follow_addon.cpp src/raidbars.cpp
  src/factions_protocol.cpp src/factions_addon.cpp
  src/saved_expeditions_protocol.cpp src/saved_expeditions_addon.cpp
  src/pet_inventory_protocol.cpp src/pet_inventory_addon.cpp)
target_sources(bastion_core PRIVATE src/firefell.cpp src/server_commands.cpp src/kinbound.cpp src/integrity.cpp
  src/twist.cpp src/fast_camp.cpp
  src/gear_score_protocol.cpp src/gear_score_state.cpp src/gear_score_addon.cpp
  src/home_city_protocol.cpp src/home_city_addon.cpp src/raid_ready_check.cpp src/expedition_journal.cpp
  src/tradeskill_plans.cpp
  src/telegraphs_protocol.cpp src/telegraphs_state.cpp src/telegraphs_geometry.cpp src/telegraphs_adapter.cpp src/telegraphs_controls.cpp)
target_compile_definitions(bastion_core PRIVATE HAILSTONE_HOST)
# The Bastion preview and public Hailstone SDKs expose different optional extensions.
foreach(extension IN ITEMS BOUNDED_SURFACE_PATCH SURFACE_CACHE_REVISION)
  file(STRINGS "${HAILSTONE_SDK_DIR}/hailstone_world_geometry.h" declaration
    REGEX "^[ \t]*HAILSTONE_WORLD_${extension}[ \t]*=" LIMIT_COUNT 1)
  if(declaration)
    target_compile_definitions(bastion_core PUBLIC "BASTION_SDK_${extension}")
  endif()
endforeach()
file(STRINGS "${HAILSTONE_SDK_DIR}/hailstone_plugin.h" _camp_dialog_api
  REGEX "\\(\\*(get_large_dialog|answer_large_dialog)\\)")
file(STRINGS "${HAILSTONE_SDK_DIR}/hailstone_plugin.h" _camping_flag
  REGEX "^#define HAILSTONE_PLAYER_CAMPING ")
list(LENGTH _camp_dialog_api _camp_dialog_count)
if(_camp_dialog_count EQUAL 2 AND _camping_flag)
  target_compile_definitions(bastion_core PUBLIC BASTION_OPTIONAL_CAMP_DIALOG=1)
endif()
set_target_properties(bastion_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(bastion_core PUBLIC src "${HAILSTONE_SDK_DIR}")
target_include_directories(bastion_core SYSTEM PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/third_party")
add_library(bastion MODULE src/plugin.cpp)
target_link_libraries(bastion PRIVATE bastion_core)
set_target_properties(bastion PROPERTIES PREFIX "" CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
if(APPLE)
  set_target_properties(bastion PROPERTIES SUFFIX ".dylib")
endif()
function(bastion_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /EHsc /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wshadow -Wconversion)
  endif()
endfunction()
bastion_warnings(bastion_core)
bastion_warnings(bastion)
add_executable(bastion_telegraph_geometry_preview tools/telegraph_geometry_preview.cpp)
target_link_libraries(bastion_telegraph_geometry_preview PRIVATE bastion_core)
bastion_warnings(bastion_telegraph_geometry_preview)
add_custom_command(TARGET bastion POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory
  "${CMAKE_CURRENT_SOURCE_DIR}/ui" "$<TARGET_FILE_DIR:bastion>/ui")
install(TARGETS bastion LIBRARY DESTINATION . RUNTIME DESTINATION .)
install(DIRECTORY ui DESTINATION .)
if(BUILD_TESTING)
  foreach(kind IN ITEMS follow_addon raidbars raid_ready_check expedition_journal tradeskill_plans factions saved_expeditions pet_inventory citadel firefell server_commands
      kinbound gear_score gear_score_addon home_city telegraphs telegraphs_geometry telegraphs_adapter integrity character_directory twist fast_camp)
    add_executable(bastion_${kind}_tests tests/${kind}_test.cpp)
    target_link_libraries(bastion_${kind}_tests PRIVATE bastion_core)
    bastion_warnings(bastion_${kind}_tests)
    target_compile_definitions(bastion_${kind}_tests PRIVATE HAILSTONE_HOST)
    add_test(NAME bastion_${kind} COMMAND bastion_${kind}_tests)
  endforeach()
  add_executable(bastion_plugin_tests tests/plugin_test.cpp src/plugin.cpp)
  target_link_libraries(bastion_plugin_tests PRIVATE bastion_core)
  bastion_warnings(bastion_plugin_tests)
  add_test(NAME bastion_plugin COMMAND bastion_plugin_tests)
  add_executable(bastion_server_integrity_tests tests/server_integrity_test.cpp)
  bastion_warnings(bastion_server_integrity_tests)
  add_test(NAME bastion_server_integrity COMMAND bastion_server_integrity_tests)
  find_package(Python3 3.8 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME bastion_legacy_integrity_approval COMMAND "${Python3_EXECUTABLE}" -B
      "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_prepare_legacy_integrity_approval.py")
  endif()
  find_program(BASTION_LUA NAMES lua5.4 lua54 lua)
  if(BASTION_LUA)
    foreach(kind IN ITEMS follow raidbars raid_manager map_filters factions saved_expeditions expedition_journal pet_inventory citadel home_city gear_score telegraphs host_adapter)
      if(NOT HAILSTONE_CLIENT_UI_ROOT AND NOT kind MATCHES "^(follow|raidbars|raid_manager|map_filters|host_adapter)$")
        continue()
      endif()
      if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/${kind}_ui_test.lua")
        add_test(NAME bastion_${kind}_ui COMMAND "${BASTION_LUA}"
          "${CMAKE_CURRENT_SOURCE_DIR}/tests/${kind}_ui_test.lua" "${CMAKE_CURRENT_SOURCE_DIR}")
        set_tests_properties(bastion_${kind}_ui PROPERTIES ENVIRONMENT "HAILSTONE_CLIENT_UI_ROOT=${HAILSTONE_CLIENT_UI_ROOT}")
      endif()
    endforeach()
    if(NOT HAILSTONE_CLIENT_UI_ROOT)
      message(STATUS "Native pack-helper UI tests require HAILSTONE_CLIENT_UI_ROOT; SDK-only and isolated UI tests remain enabled")
    endif()
  endif()
endif()
